def rolling_mean(values, width):
    if width <= 0:
        raise ValueError("width must be positive")
    total = 0.0
    result = []
    for index, value in enumerate(values):
        total += value
        if index >= width:
            total -= values[index - width]
        if index + 1 >= width:
            result.append(total / width)
    return result
